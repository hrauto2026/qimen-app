// 🌟 收集此宮所有干支並執行合沖刑害高亮 let activeStems = new Set(); let activeBranchList = []; // 1. 收集天干（天盤干、地盤干） if (data.hStems) data.hStems.forEach(s => activeStems.add(s)); if (data.eStems) data.eStems.forEach(s => activeStems.add(s)); // 2. 收集宮位本身地支（宮支放最前） let palaceBranches = []; if (data.branch) { palaceBranches = data.branch.split('/').filter(Boolean); palaceBranches.forEach(b => { activeBranchList.push({ char: b, display: b, source: 'palace' }); }); } // 3. 收集 休/傷/景/驚門 所帶之地支（門支） const DOOR_BRANCH_MAP = { '休門': { branch: '子', label: '休' }, '傷門': { branch: '卯', label: '傷' }, '景門': { branch: '午', label: '景' }, '驚門': { branch: '酉', label: '驚' } }; if (data.door && DOOR_BRANCH_MAP[data.door]) { const dInfo = DOOR_BRANCH_MAP[data.door]; if (!palaceBranches.includes(dInfo.branch)) { activeBranchList.push({ char: dInfo.branch, display: `${dInfo.branch}（${dInfo.label}）`, source: 'door' }); } } // 4. 收集天盤干遁支 if (data.hStems) { data.hStems.forEach(s => { if (DUN_GAN_MAP[s]) { let b = DUN_GAN_MAP[s]; activeBranchList.push({ char: b, display: `${b}（天${s}）`, source: 'hStem' }); } }); } // 5. 收集地盤干遁支 if (data.eStems) { data.eStems.forEach(s => { if (DUN_GAN_MAP[s]) { let b = DUN_GAN_MAP[s]; activeBranchList.push({ char: b, display: `${b}（地${s}）`, source: 'eStem' }); } }); } // 6. 渲染合沖刑害列表並更新徽章 const { html: relationHtml, totalMatches } = renderRelationBox(activeStems, activeBranchList); document.getElementById('pm-cheatsheet-content').innerHTML = relationHtml;
